async function crearProducto() { const token = localStorage.getItem("token"); const fileInput = document.getElementById("imagen"); // 1. Validar que se haya seleccionado una imagen if (fileInput.files.length === 0) { alert("Por favor, selecciona una imagen"); return; } try { // 2. Primero subimos la imagen para obtener la URL const formData = new FormData(); formData.append("imagen", fileInput.files[0]); const resImg = await fetch("https://ventas-backend-wj4v.onrender.com/subir-imagen", { method: "POST", body: formData // Nota: No incluyas Content-Type en los headers al usar FormData, // el navegador lo hace automáticamente con el boundary correcto. }); const dataImg = await resImg.json(); const urlImagen = dataImg.url; // Asumiendo que el backend devuelve { url: "..." } // 3. Determinar el endpoint según el estado const tipoSeleccionado = document.getElementById("estado").value; const endpointMap = { nuevo: "productos-nuevos", usado: "productos-usados", servicio: "productos-servicios" }; const endpoint = endpointMap[tipoSeleccionado]; // 4. Crear el producto con la URL de la imagen recibida const res = await fetch(`https://ventas-backend-wj4v.onrender.com/${endpoint}`, { method: "POST", headers: { "Content-Type": "application/json", "Authorization": `Bearer ${token}` }, body: JSON.stringify({ nombre: document.getElementById("titulo").value, marca: document.getElementById("marca").value, cantidad: Number(document.getElementById("cantidad").value), precio: Number(document.getElementById("precio").value), condicion: tipoSeleccionado, imagen: [urlImagen], // Guardamos la URL real, no el path local vendedor: "Nombre visible", descripcion: document.getElementById("descripcion").value, ubicacion: { calle: document.getElementById("calle").value, altura: document.getElementById("altura").value, localidad: document.getElementById("localidad").value }, direcc: `../vistas/${tipoSeleccionado}.html` }) }); if (res.ok) { alert("Producto creado exitosamente"); } else { alert("Error al crear el producto"); } } catch (error) { console.error("Error en el proceso:", error); alert("Hubo un fallo en la conexión"); } } function salir() { localStorage.removeItem("token"); location.href = "login.html"; }
